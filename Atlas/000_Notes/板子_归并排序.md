void merge_sort(int l, int r) {
  if (l == r) return;  // 递归终止
  int mid = (l + r) >> 1;
  // 下面两行是不断的划分元素的过程，通过递归实现从最小的元素开始处理，完事return返回上个函数调用栈，逐渐扩大处理的范围，也是很巧妙。
  merge_sort(l, mid); // 不断切割左区间
  merge_sort(mid + 1, r);  // 不断切割右区间

  // 开始归并排序，注意，这里通过mid来划分左右归并段，并不是开两个辅助数组
  int i = l, j = mid + 1, k = l; // 这里的i注意是从l开始，不是0！
  while (i <= mid && j <= r) {
    if (a[i] <= a[j])
      b[k++] = a[i++];
    else
      b[k++] = a[j++];
  }
  while (i <= mid) b[k++] = a[i++];
  while (j <= r) b[k++] = a[j++];

  for (int i = l; i <= r; i++) a[i] = b[i]; // 这里的i注意是从l开始，不是0！
}
