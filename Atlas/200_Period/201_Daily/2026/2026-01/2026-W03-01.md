// ============================================
// 本周聚焦项目查询
// 查询指定时间段内更新的项目笔记
// ============================================
// --- 配置区 ---
const start = dv.current().start_week;    // 本周开始日期
const end = dv.date(dv.current().end_week) // 本周结束日期
    .plus({ days: 1 });                    // 调整为包含结束日
// --- 查询逻辑 ---
// 从指定文件夹中筛选符合条件的项目
const pages = dv.pages('"40_Permanent" OR "30_Projects"')
    .where(p => 
        p.type == "project" &&           // 类型为项目
        p.file.mtime >= start &&         // 修改时间在开始日期之后
        p.file.mtime <= end              // 修改时间在结束日期之前
    )
    .sort(p => p.file.mtime, "desc");    // 按修改时间降序排列
// --- 渲染逻辑 ---
if (pages.length > 0) {
    // 显示标题
    dv.header(2, "本周聚焦项目");
    // 创建表格展示项目信息
    dv.table(
        ["项目", "当前状态", "领域", "最近更新"], 
        pages.map(p => [
            p.file.link,    // 项目文件链接
            p.status,       // 项目状态
            p.area,         // 项目领域
            p.file.mtime    // 最近更新时间
        ])
    );
} else {
    // 可选：显示无结果的提示信息
    dv.paragraph("本周没有更新的项目。");
}

// 本周任务完成情况查询
// 查询指定时间段内完成的任务
// ============================================
// --- 配置区 ---
const start = dv.current().start_week;    // 本周开始日期
const end = dv.date(dv.current().end_week) // 本周结束日期
    .plus({ days: 1 });                    // 调整为包含结束日
// --- 查询逻辑 ---
// 注意：这里扫描了所有页面。如果你的库很大，建议将 dv.pages() 范围缩小，比如 dv.pages('"30_Projects"')
const tasks = dv.pages().file.tasks
    .where(t => 
        t.completed &&                 // 任务已完成
        t.completion >= start &&      // 完成时间在开始日期之后
        t.completion <= end           // 完成时间在结束日期之前
    );
// --- 渲染逻辑 ---
if (tasks.length > 0) {
    // 显示标题
    dv.header(2, "本周任务完成情况");
    
    // 以任务列表形式展示完成的任务
    dv.taskList(tasks);
} else {
    // 可选：显示无结果的提示信息
    dv.paragraph("本周没有完成的任务。");
}
